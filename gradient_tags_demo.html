同列標籤彩虹漸層分佈示範模板 (Rainbow Gradient Tags Template)

🌙 暗黑模式
🎨 漸層設定與即時預覽控制項
💡 原理說明：透過 JavaScript 取得表格中目前可視的總列數，依據行索引值 (Row Index) 均勻分配 HSL 色碼的 Hue (色相) 值。不論表格有多少列，標籤色彩都會從上到下呈現滑順無縫的彩虹漸層過渡。

💻 開發者接入指南與程式碼範例

在渲染表格後執行此段程式，它會獲取所有的行，並依據其行索引從上到下為指定的標籤元素動態注入彩虹漸層 HSL 樣式：

javascript 
function applyRainbowGradient() {
  const rows = document.querySelectorAll('#scheduleTable tbody tr');
  const totalRows = rows.length;
  if (totalRows === 0) return;

  // 取得 UI 面板所設定的風格與範圍參數
  const colTarget = document.getElementById('columnSelect').value;
  const styleMode = document.getElementById('colorStyle').value;
  const hueRange = document.getElementById('hueRange').value;
  const isDark = document.documentElement.getAttribute('data-theme') === 'dark';

  // 決定彩虹的色相 (Hue) 起點與終點
  let startHue = 0;   // 紅色
  let endHue = 300;   // 紫紅色
  if (hueRange === 'cool') { startHue = 120; endHue = 270; }
  else if (hueRange === 'warm') { startHue = -30; endHue = 45; }
  else if (hueRange === 'soft') { startHue = 240; endHue = 330; }

  rows.forEach((row, index) => {
    // 依據當前列的 index 比例，計算出對應的 HSL Hue 色相
    const ratio = totalRows > 1 ? index / (totalRows - 1) : 0;
    let hue = startHue + ratio * (endHue - startHue);
    if (hue < 0) hue += 360; // 確保色相在 0-360 區間

    // 定義各個要套用的欄位標籤類別
    const targets = {
      leader: row.querySelector('.tag-leader'),
      speaker: row.querySelector('.tag-speaker'),
      status: row.querySelector('.tag-status')
    };

    // 針對選定的目標套用色彩
    for (const [key, tagEl] of Object.entries(targets)) {
      if (!tagEl) continue;

      // 判斷是否為目前選中的目標列 (或全部)
      const shouldApply = (colTarget === 'all' || colTarget === key);

      if (shouldApply) {
        // 套用動態 HSL
        setTagStyle(tagEl, hue, styleMode, isDark);
      } else {
        // 重設為預設灰色調
        tagEl.style.backgroundColor = isDark ? '#334155' : '#f1f5f9';
        tagEl.style.color = isDark ? '#cbd5e1' : '#475569';
        tagEl.style.borderColor = 'transparent';
      }
    }
  });
}

function setTagStyle(element, hue, mode, isDark) {
  if (mode === 'pastel') {
    // 馬卡龍雅色：背景極淺 飽和度高，文字深色 飽和度更高
    const bgL = isDark ? '15%' : '95%';
    const textL = isDark ? '80%' : '25%';
    const borderL = isDark ? '30%' : '85%';
    
    element.style.backgroundColor = `hsla(${hue}, 85%, ${bgL}, 1)`;
    element.style.color = `hsla(${hue}, 85%, ${textL}, 1)`;
    element.style.borderColor = `hsla(${hue}, 80%, ${borderL}, 1)`;
    element.style.borderStyle = 'solid';
    element.style.borderWidth = '1px';
  } else if (mode === 'outline') {
    // 線框色：背景透明，外框明顯，文字顏色與外框相同
    const borderL = isDark ? '60%' : '50%';
    element.style.backgroundColor = 'transparent';
    element.style.color = `hsla(${hue}, 80%, ${borderL}, 1)`;
    element.style.borderColor = `hsla(${hue}, 80%, ${borderL}, 0.8)`;
    element.style.borderStyle = 'solid';
    element.style.borderWidth = '1.5px';
  } else if (mode === 'vibrant') {
    // 亮眼色：飽和背景，文字均為純白
    element.style.backgroundColor = `hsla(${hue}, 75%, 50%, 1)`;
    element.style.color = '#ffffff';
    element.style.borderColor = 'transparent';
  }
}

CSS 樣式僅需設定基礎的圓角、內縮以及在 Outline 風格時所需的過渡動畫，具體色彩將交由 JS 動態變更：

css 
/* 標籤基礎樣式設定 */
.rainbow-tag {
  display: inline-block;
  padding: 5px 12px;
  border-radius: 30px;
  font-size: 13px;
  font-weight: 700;
  text-align: center;
  transition: all 0.3s cubic-bezier(0.4, 0, 0.2, 1);
  box-shadow: 0 2px 5px rgba(0, 0, 0, 0.03);
}

/* 確保表格列有淡入的感覺 */
.animated-row {
  animation: fadeInUp 0.4s cubic-bezier(0.4, 0, 0.2, 1) forwards;
}

@keyframes fadeInUp {
  from {
    opacity: 0;
    transform: translateY(10px);
  }
  to {
    opacity: 1;
    transform: translateY(0);
  }
}
